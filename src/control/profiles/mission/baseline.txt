# МИССИЯ И ПИЛОТ — ручки полёта вне стабилизаторов: эталон 2026-09-07 (то, чем летает cmd/bl).
# До этого дня эти ключи жили в трёх местах: freefly_lv.sh (BS_PILOT/BS_MISSION), env.default
# (BS_SF_MASTER/BS_LAND_JOY) и дефолты ноды (всё остальное — молча). Здесь они собраны ЯВНО
# теми же значениями; поведение прогона не меняется (A/B: мета lv2_joy_20260906_231055).
# Формат — README.md. Аргументы ПРОГОНА (не параметры) сюда не входят: BS_REPLAY_SCENARIO /
# BS_REPLAY_RAW / BS_REPLAY_FENCE — путь к сценарию и его забор задаются в команде запуска
# (см. mission/replay.txt).

# слой: env (freefly_lv.sh) — источник стиков и плейлист. joy = живой пульт TX12 по USB
# (joy_linux_node → /joy); freefly = «свободный полёт»: взлёт руками, ярусы лесенки по
# CH6/SF, посадка по SA. Реплей сценария — mission/replay.txt.
BS_PILOT=joy
BS_MISSION=freefly

# слой: env (env.default) — SF (CH7) = мастер сырых стиков, SC (CH6) = потолок лесенки
# демпфер/VinsHold/LOITER (sf-master-ladder); нужен микс SF→CH7 в EdgeTX
BS_SF_MASTER=1
# кнопка SA (мягкая посадка SoftLand) в /joy пульта: b<i>/a<i>. ⚠️ ЗАВИСИТ ОТ МИКШЕРА
# ПУЛЬТА — другой пульт/прошивка → перемерить src/lab/joystick/js_probe.py
BS_LAND_JOY=b1

# кнопка SD (ВОЗВРАТ ДОМОЙ) в /joy пульта: b<i>/a<i>, '' = кнопки нет (возврат только
# `make rth` / `make smart-rth`). Фронт нажатия = тот же импульс, что от топика: в freefly
# прыжок на шаг rth, повторное нажатие ОТМЕНЯЕТ возврат. Режим — BS_RTH_JOY_MODE.
# ⚠️ ЗАВИСИТ ОТ МИКШЕРА: SD должен быть назначен на канал ПОСЛЕ осей (CH11 = b2 по коду
# EdgeTX classic, CH9..CH32 = b0..b23); перемерить src/lab/joystick/js_probe.py
BS_RTH_JOY=b2
BS_RTH_JOY_MODE=SMART_RTL

# ЛАТЧ ДОВЕРИЯ К ВОЗВРАТУ (rth_ready.py): в круге радиусом BS_RTH_RADIUS (метры ПО IPM;
# 2026-09-09 поднят 5 → 10 — больше запаса на созревание VINS. ⚠️ ЦЕНА: дом идёт за позой
# до первого выхода из круга, значит садиться борт будет там, где границу пересёк, а по
# IPM 10 м = 13–20 м физически (канал занижает с гейном 0.5–0.77))
# от точки арма VINS может перерождаться сколько угодно — лечимся; как только он зрел и
# здоров BS_RTH_RIPE_SEC подряд, латчим дом (поза EKF) и начинаем писать трек шагом
# BS_RTH_TRACK_M. Не успели за BS_RTH_HEAL_SEC или ушли из круга — возврата в этом полёте
# нет (rth=lost в статусе, красный баннер в FPV). Та же зрелость открывает мост в EKF.
BS_RTH_RADIUS=10
BS_RTH_HEAL_SEC=60
BS_RTH_RIPE_SEC=5
BS_RTH_RIPE_N=100
BS_RTH_TRACK_M=3
BS_RTH_JUMP_M=2
# дом ставим ОДИН раз, как только рама устоялась: ждём столько секунд спокойной позы EKF
# после открытия моста (там полётник скачком пересаживается на нашу раму — ray_tracer
# отдаёт сырой VINS 2 с). Скачок в выдержке перезапускает отсчёт
BS_RTH_HOME_SETTLE=3
# зрелость VINS дополнительно судится по УСТОЙЧИВОСТИ разности курсов «AHRS − VINS»
# (brdn= в статусе): у сошедшегося кадра она стоит. 0 = чек выкл
# ⚠️ ВЫКЛЮЧЕН 2026-09-10: допуск стоял НА УРОВНЕ СОБСТВЕННОГО ШУМА сигнала.
# Замер размаха brdn в скользящем окне 3 с: 164041 медиана 2.6° (51 % окон выше
# допуска, дом залатчен), 184108 медиана 2.2° (39 %, залатчен), yawab_check3
# медиана 10.6° (95 %, дом НЕ залатчен — rth=lost:timeout, мост закрыт весь
# полёт по ripe). Шум есть и при |ω|=0, так что гейт виража (BS_RTH_DYAW_WZ)
# от него не спасает. Латч проскакивал на тихих участках — то есть чек работал
# лотереей и однажды запретил возврат наглухо, нарушив собственное правило:
# чек имеет право ОТЛОЖИТЬ зрелость, но не запретить полёт.
# Оживлять — только с допуском ВЫШЕ измеренного шума (≥15°) или по фильтрованному
# Δyaw; но при 15° он ловит лишь грубый развал рамы, а его и так ловит детектор
# перерождения. Механика и тесты (test_rth_ready §10д/10ж/10з) оставлены на месте.
BS_RTH_DYAW_TOL=0
BS_RTH_DYAW_WZ=15

# ЧЕЙ КУРС ДЕРЖИТ EKF: compass (EK3_SRC1_YAW=1, как всегда) | vins (после латча
# возврата переключаем на 6 = ExternalNav). Дефолт compass, пока не доказано полётом;
# механика, риски и порядок проверки — src/nav/frames.md
BS_EKF_YAW_SRC=compass
BS_EKF_YAW_FALLBACK=1
BS_RTH_SET_HOME=1

# КАК возвращаемся: guided = сами, по своему треку (GUIDED + поток уставок, курс по
# треку, по прибытии сразу мягкая посадка); fcu = режимом полётника (RTL/SMART_RTL,
# BS_RTH_JOY_MODE). Радиус приёмки точки и ожидаемая скорость (последняя — только для
# бюджета шага, реальную держит WP_SPD полётника).
BS_RTH_MODE=guided
BS_RTH_WP_R=1.5
BS_RTH_SPEED=2
BS_RTH_LAND_HOME=1

# guard возврата: шаг rth сам отменяет возврат, если мост VINS→EKF закрыт или гейт
# здоровья объявил VINS больным (в LV=2 EKF держится только нашей подтяжкой — закрылся
# мост, и борт летит домой по цифрам, которых больше нет; разбор 044105 в cmd/smart_rth)
BS_RTH_GUARD=1

# слой: нода — пульт: устройство (bootstrap_arch2.sh, ⚠️ машинно-зависимо) и знаки осей

BS_JOY_DEV=/dev/input/js0
BS_JOY_SIGNS=
# слой: нода (2026-09-07 — до этого без проводки BS_): мёртвая зона и полный ход стика

BS_PILOT_DEADZONE=30
BS_PILOT_FULL=400
BS_PILOT_PITCH_SIGN=-1
BS_PILOT_ROLL_SIGN=1

# слой: нода — ВЫСОТА: целевая (скриптовые миссии) и внешний контур AltHold

# слой: нода (2026-09-07 — до этого без проводки BS_, только дефолт в коде): пересчёт

BS_ALT_DZ=100
BS_ALT_SPAN=400

# слой: нода — бюджеты фаз, sim-с (mode_budget — в loiter/: латч режима 80 с)
BS_ARM_BUDGET=40
BS_CLIMB_BUDGET=60
BS_LAND_BUDGET=45

# слой: нода — ГЕОЗАБОР, м от точки старта (0 = выкл): ушли дальше — сегмент прерывается,
