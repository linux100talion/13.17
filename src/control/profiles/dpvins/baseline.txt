# скорости + внешний √-кап позиции при отпущенном стике, трим ветра в осях мира. Скорость

BS_VINS_STAB=dpvins

# слой: нода — ГЕЙНЫ (кампания ab_dpv_* 2026-09-03: kp 80→40 убрал предельный цикл
# σθ 1.05°→0.5°). kp — PWM на м/с ошибки скорости (вперёд/вбок); ki — рабочий трим после
# первого гвоздя; ki_trim — до гвоздя (быстрый захват ветра: при ветре 10 ≈ 100 PWM,
# ki 6 → унос 17 м, ki_trim 60 → ~2); imax — потолок трима; cmd_gain — м/с при полном
# стике; vsmooth — ФНЧ скорости внутреннего контура (τ, с).
BS_DPVINS_KP_FWD=40

BS_DPVINS_VSMOOTH=0.3
# внешний позиционный контур (стик отпущен): возврат к гвоздю pos_kp, кап vmax, √-разгон acc
BS_DPVINS_POS_KP=0.3
BS_DPVINS_POS_VMAX=0.3
BS_DPVINS_POS_ACC=0.15
# слой: нода — ФАЗА BRAKE станции (закон демпфера, 2026-09-05): 0 = выкл в эталоне.

# гвоздь на входе: 0 = только по стопу после движения (эталон), 1 = сразу, если трим посеян и стоим (кандидат brake5_pin)
