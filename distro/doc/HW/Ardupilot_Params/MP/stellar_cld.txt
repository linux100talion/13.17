# === Рама / моторы =============================================================
FRAME_CLASS,1            # квадрик (заводское 0 = рама не задана, моторных выходов нет)
FRAME_TYPE,1             # X (совпадает с заводским, фиксируем явно)
# Карта выходов снята тестом моторов 2026-09-23 (PDB развёрнута относительно рамы):
#   выход 1 = передний правый, 2 = передний левый, 3 = задний правый, 4 = задний левый.
# Схема ArduPilot Quad X (props-in): M1 FR ↺, M2 RL ↺, M3 FL ↻, M4 RR ↻ (вид сверху).
SERVO1_FUNCTION,33       # выход 1 = мотор 1 (передний правый)
SERVO2_FUNCTION,35       # выход 2 = мотор 3 (передний левый)
SERVO3_FUNCTION,36       # выход 3 = мотор 4 (задний правый)
SERVO4_FUNCTION,34       # выход 4 = мотор 2 (задний левый)
MOT_PWM_TYPE,5           # DShot300 (как в старом дампе); заводское 0 = обычный PWM
SERVO_BLH_AUTO,1         # проброс BLHeli к ESC на всех моторных выходах (конфигуратор по USB)
# SERVO_BLH_PORT = 0 (USB) — заводское, не трогаем
#
# ESC: 4 шт. Flycolor X-CROSS HV3 60A 5-12S на лучах, ПО BLHeliSuite32 (distro/doc/HW/hw.txt).
#   ⚠ «E-X-30 Bluejay 0.21.1» из esc-configurator-log.txt — ДРУГОЙ борт (FC ID 0x01f0…,
#   у нашего 0x0260…). Проброс ArduPilot + esc-configurator 2026-09-23 не читает; для
#   BLHeli_32 штатный путь — BLHeliSuite32 через проброс ArduPilot.
SERVO_DSHOT_ESC,1        # BLHeli_32/AM32: DShot-команды (бипер, реверс); с ним пошли обороты ESC_TELEMETRY (2 = Bluejay — телеметрии не было)
SERVO_BLH_BDMASK,15      # двунаправленный DShot на моторах 1-4 (hwdef: BIDIR на TIM3/TIM5) — обороты для notch
SERVO_BLH_TMOUT,10       # выход из проброса к ESC через 10 с тишины (0 = никогда: FC застревал без MAVLink до снятия питания)
SERVO_BLH_RVMASK,8       # реверс DShot на выходе 4 (задний левый крутился ↻, нужно ↺); остальные три по схеме
#
# ЖДУТ теста моторов на нашей раме (старые значения — с другой настройки/пропов):
# MOT_SPIN_ARM,0.04      # обороты при арме; заводское 0.10
# MOT_SPIN_MIN,0.07      # мин. в полёте, должно быть ≥ SPIN_ARM + 0.03; заводское 0.15
# MOT_SPIN_MAX,0.6       # ⚠ старое ограничивало газ 60 % — выяснить зачем; заводское 0.95
# MOT_THST_EXPO,0.2      # кривая тяги, зависит от пропов; заводское 0.65 (5")
# MOT_THST_HOVER — не задаём: ArduPilot учит сам (MOT_HOVER_LEARN)

# === Батарея 8S4P Molicel P45B, 18000 мА·ч (distro/doc/HW/hw.txt) =================
# hwdef StellarH7V2: пин 10 = делитель на плате FC (×11), пин 18 = делитель PDB (×21).
# Разъём батареи — на PDB → основной монитор на пин 18. Старый дамп читал пин 10 ×21 —
# на этой прошивке там 0.59 В сырых (≈12 В), не батарея.
BATT_VOLT_PIN,18         # делитель PDB
BATT_VOLT_MULT,21.26     # калибровка 2026-09-23: мультиметр 32.40 В на PDB / FC 32.01 В при ×21 (номинал 20k/1k)
BATT_AMP_PERVLT,40       # из старого дампа (заводское 66.7); ПРОВЕРИТЬ под нагрузкой
BATT_CAPACITY,18000
BATT_LOW_VOLT,26         # 3.25 В/банку (как в старом)
BATT_CRT_VOLT,24         # 3.0 В/банку — посадка по hw.txt (как в старом)
BATT_FS_LOW_ACT,2        # low → RTL (без позиции ArduPilot сядет); в старом 0 = ничего — не переносим
BATT_FS_CRT_ACT,1        # critical → Land
BATT2_MONITOR,0          # второй монитор не нужен (заводское 4 читало пин 18, пока основной смотрел в пин 10)
